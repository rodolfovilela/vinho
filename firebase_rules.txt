rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

	 match /bookings/{userId} {
        allow read: if true;

        allow create: if request.auth != null
                      && request.auth.uid == userId
                      && isValidBooking();

        allow update, delete: if false;
      }
      
      match /searchLogs/{docId} {
        allow read	: if false;

        allow write: if request.resource.data.lastSearchTimestamp is timestamp &&
    request.resource.data.count is int;

        allow delete: if false;
      }
      
      
      
       match /searchLogs/{docId}/timestamps/{timestampId} {
       allow create: if request.resource.data.keys().hasOnly(['timestamp']) &&
                request.resource.data.timestamp is timestamp;
      allow delete,read: if false;
    }
      
      match /exceptions/{document=**} {
        allow write: if true;

        allow update, delete, read: if false;
      }
    
    match /{document=**} {
      allow read: if true;
      allow write: if false;
    }
  }
}

function isValidBooking() {
  return request.resource.data.keys().hasOnly([
    'userId',
    'name',
    'email',
    'phone',
    'seats',
    'createdAt'
  ])
  && request.resource.data.userId == request.auth.uid
  && request.resource.data.name is string
  && request.resource.data.email is string
  && request.resource.data.phone is string
	&& request.resource.data.seats is int
  && request.resource.data.createdAt is timestamp;
}
